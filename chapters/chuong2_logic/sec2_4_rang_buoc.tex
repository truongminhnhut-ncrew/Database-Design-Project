\section{Thiết kế ràng buộc toàn vẹn mức logic}
\label{sec:2_4_rang_buoc_logic}

\subsection{Giới thiệu và Căn cứ Phương pháp luận}
Trong chu kỳ sống của một Hệ quản trị Cơ sở dữ liệu (DBMS), giai đoạn \textbf{Thiết kế Logic} đóng vai trò là cầu nối cốt lõi giữa Mô hình Quan niệm (ERD/EER/UML) và Thiết kế Vật lý trên một hệ quản trị cụ thể (PostgreSQL). Mục tiêu của phần này là thực hiện ánh xạ chuẩn hóa các thực thể, thuộc tính và mối kết hợp từ bài toán \textbf{FamilyConnect: Nền tảng Cộng đồng Gia đình số tích hợp Trí tuệ Nhân tạo} sang Lược đồ Quan hệ (Relational Schema).

Đồng thời, nội dung tập trung phân tích triệt để các \textbf{Phụ thuộc hàm (FD)}, tính toán bao đóng, xác định khóa ứng viên và tiến hành chuẩn hóa lược đồ đạt \textbf{Dạng chuẩn Boyce-Codd (BCNF)} hoặc \textbf{Dạng chuẩn 3 (3NF)} nhằm loại bỏ hoàn toàn các bất thường dữ liệu (biến động thêm, sửa, xóa) và thiết lập hệ thống \textbf{Ràng buộc Toàn vẹn (Integrity Constraints)} mức logic vững chắc.

\subsection{Kiến trúc Chi tiết và Lược đồ Quan hệ Logic}

Dựa trên các yêu cầu nghiệp vụ của hệ thống \textbf{FamilyConnect}, hệ thống được phân rã thành các phân hệ chính bao gồm:
\begin{enumerate}[label=\arabic*.]
    \item \textbf{User \& Security}: Quản lý người dùng, tài khoản, vai trò (RBAC) và xác minh thành viên.
    \item \textbf{Family \& Genealogy}: Quản lý gia đình, chi/nhánh/phái, thành viên, quan hệ cha mẹ - con cái và hôn nhân.
    \item \textbf{Community \& Events}: Quản lý bài đăng, tương tác, tin tức, sự kiện, phản hồi RSVP và lưu trữ di sản số.
    \item \textbf{AI-Assisted Services \& Heritage}: Tìm kiếm ngữ nghĩa bằng AI, nhật ký trợ lý AI, lưu trữ văn bản di sản và tài liệu lịch sử.
    \item \textbf{System Administration}: Kiểm duyệt, nhật ký kiểm toán (Audit Log) và cấu hình.
\end{enumerate}

\subsubsection{Danh sách các Lược đồ Quan hệ Ánh xạ từ ERD}
Dưới đây là tập hợp các quan hệ logic thu được sau khi áp dụng Thuật toán ánh xạ ER/EER sang Mô hình Quan hệ:

\begin{enumerate}[label=\textbf{\arabic*.}]
    \item \textbf{Users} (\underline{user\_id}, username, email, password\_hash, full\_name, phone\_number, created\_at, is\_active, role\_code)
    \item \textbf{FamilyTree} (\underline{family\_id}, family\_name, founding\_date, description, created\_by\_user\_id\textsuperscript{FK})
    \item \textbf{Branch} (\underline{branch\_id}, branch\_name, generation\_level, description, family\_id\textsuperscript{FK})
    \item \textbf{MemberProfile} (\underline{member\_id}, first\_name, last\_name, gender, birth\_date, death\_date, is\_alive, bio, occupation, education\_level, current\_location, user\_id\textsuperscript{FK}, branch\_id\textsuperscript{FK})
    \item \textbf{ParentChildRelation} (\underline{parent\_id\textsuperscript{FK}, child\_id\textsuperscript{FK}}, relation\_type)
    \item \textbf{MarriageRelation} (\underline{marriage\_id}, husband\_id\textsuperscript{FK}, wife\_id\textsuperscript{FK}, marriage\_date, divorce\_date, status)
    \item \textbf{Post} (\underline{post\_id}, title, content, visibility\_scope, created\_at, author\_member\_id\textsuperscript{FK}, family\_id\textsuperscript{FK})
    \item \textbf{PostInteraction} (\underline{interaction\_id}, post\_id\textsuperscript{FK}, member\_id\textsuperscript{FK}, reaction\_type, comment\_text, created\_at)
    \item \textbf{Event} (\underline{event\_id}, event\_title, event\_description, start\_time, end\_time, location, event\_type, family\_id\textsuperscript{FK}, organizer\_member\_id\textsuperscript{FK})
    \item \textbf{EventRSVP} (\underline{event\_id\textsuperscript{FK}, member\_id\textsuperscript{FK}}, rsvp\_status, updated\_at)
    \item \textbf{HeritageDocument} (\underline{document\_id}, title, document\_type, file\_path, historical\_period, summary, uploaded\_by\_member\_id\textsuperscript{FK}, family\_id\textsuperscript{FK})
    \item \textbf{AIAssistantLog} (\underline{log\_id}, member\_id\textsuperscript{FK}, prompt\_query, response\_summary, vector\_embedding\_id, logged\_at)
    \item \textbf{AuditLog} (\underline{audit\_id}, user\_id\textsuperscript{FK}, action\_type, entity\_name, entity\_id, timestamp, ip\_address)
\end{enumerate}

\subsection{Phân tích Phụ thuộc Hàm và Chuẩn hóa Lược đồ}

Để đảm bảo cơ sở dữ liệu không bị lặp lại thông tin, tiết kiệm không gian lưu trữ và tránh các dị thường dữ liệu (Anomalies) khi cập nhật, chúng ta áp dụng lý thuyết Phụ thuộc hàm (Functional Dependency - FD) để phân tích và tiến hành chuẩn hóa.

\subsubsection{Xác định Tập Phụ thuộc Hàm ($F$) cho các Quan hệ Trọng yếu}

\begin{example}[\textbf{Phân tích Phụ thuộc hàm cho Quan hệ MemberProfile}]
Xét lược đồ quan hệ $MemberProfile(U)$ với tập thuộc tính:
$$U = \{\text{member\_id (M)}, \text{user\_id (U)}, \text{email (E)}, \text{branch\_id (B)}, \text{branch\_name (BN)}, \text{generation\_level (G)}\}$$

Tập phụ thuộc hàm $F$ bao gồm:
\begin{align*}
    f_1: & \quad M \rightarrow \{U, B, BN, G\} \\
    f_2: & \quad U \rightarrow E \\
    f_3: & \quad B \rightarrow \{BN, G\}
\end{align*}

\paragraph{Tính Bao đóng và Tìm Khóa Ứng viên:}
Tính bao đóng của $\{M\}^+$ theo tập phụ thuộc hàm $F$:
\begin{itemize}
    \item Bước 1: Khởi tạo $\{M\}^+ = \{M\}$.
    \item Bước 2: Áp dụng $f_1 \implies \{M\}^+ = \{M, U, B, BN, G\}$.
    \item Bước 3: Áp dụng $f_2$ (vì $U \subseteq \{M\}^+$) $\implies \{M\}^+ = \{M, U, B, BN, G, E\} = U$.
\end{itemize}
Do $\{M\}^+ = U$ và không có tập con thực sự nào của $\{M\}$ xác định được $U$, nên $K = \{M\}$ là \textbf{Khóa ứng viên duy nhất} (và là Khóa chính) của lược đồ $MemberProfile$.
\end{example}

\subsubsection{Kiểm tra Dạng chuẩn và Phân rã Lược đồ}

Xét lược đồ $MemberProfile$ ở trên, ta nhận thấy:
\begin{itemize}
    \item Khóa chính là $M$.
    \item Tồn tại phụ thuộc hàm $f_3: B \rightarrow \{BN, G\}$. Thuộc tính $B$ (mã chi/nhánh) không phải là siêu khóa, và $\{BN, G\}$ không phải là thuộc tính nguyên tố.
    \item Tồn tại phụ thuộc bắc cầu $M \rightarrow B \rightarrow \{BN, G\}$.
\end{itemize}
$\implies$ Lược đồ $MemberProfile$ \textbf{vi phạm Dạng chuẩn 3 (3NF)} và \textbf{Dạng chuẩn Boyce-Codd (BCNF)}. Điều này dẫn đến dị thường dữ liệu: khi tên chi/nhánh hoặc cấp thế hệ của $B$ thay đổi, ta phải cập nhật trên tất cả các hồ sơ thành viên thuộc chi đó.

\paragraph{Phân rã đảm bảo Bảo toàn Phụ thuộc và Phép kết Không tổn thất (Lossless Join):}
Áp dụng thuật toán phân rã BCNF, ta tách $MemberProfile$ thành 2 quan hệ con:
\begin{enumerate}[label=\arabic*.]
    \item $Branch\_Schema(\underline{branch\_id}, branch\_name, generation\_level)$ với tập FD $F_1 = \{B \rightarrow \{BN, G\}\}$ đạt BCNF.
    \item $MemberProfile\_Schema(\underline{member\_id}, user\_id, branch\_id\textsuperscript{FK})$ với tập FD $F_2 = \{M \rightarrow \{U, B\}, U \rightarrow E\}$ đạt BCNF.
\end{enumerate}

\subsection{Ràng buộc Toàn vẹn Mức Logic (Integrity Constraints)}

Ràng buộc toàn vẹn mức logic là các quy tắc nghiệp vụ bắt buộc hệ quản trị CSDL phải tuân thủ nhằm đảm bảo tính đúng đắn, nhất quán của dữ liệu. Hệ thống \textbf{FamilyConnect} áp dụng 4 nhóm ràng buộc toàn vẹn chính:

\subsubsection{Ràng buộc Miền giá trị (Domain Constraints)}
\begin{itemize}
    \item \textbf{C1.1 (Gender Domain)}: Giá trị thuộc tính $gender \in \{\text{'Male'}, \text{'Female'}, \text{'Other'}\}$.
    \item \textbf{C1.2 (RSVP Status Domain)}: $rsvp\_status \in \{\text{'Attending'}, \text{'Declined'}, \text{'Pending'}, \text{'Maybe'}\}$.
    \item \textbf{C1.3 (Reaction Type Domain)}: $reaction\_type \in \{\text{'Like'}, \text{'Love'}, \text{'Celebrate'}, \text{'Care'}\}$.
\end{itemize}

\subsubsection{Ràng buộc Thực thể và Khóa (Entity \& Key Constraints)}
\begin{itemize}
    \item \textbf{C2.1 (Primary Keys)}: Tất cả các bảng phải có Khóa chính (Primary Key) xác định duy nhất từng bản ghi và không được phép nhận giá trị \texttt{NULL}.
    \item \textbf{C2.2 (User Unique Identity)}: $username$ và $email$ trong bảng \texttt{Users} phải là duy nhất (\texttt{UNIQUE}).
    \item \textbf{C2.3 (Marriage Uniqueness)}: Cặp $(husband\_id, wife\_id, marriage\_date)$ trong quan hệ \texttt{MarriageRelation} là duy nhất.
\end{itemize}

\subsubsection{Ràng buộc Toàn vẹn Tham chiếu (Referential Integrity Constraints)}
\begin{itemize}
    \item \textbf{C3.1 (Member to User Reference)}:
    $$\forall m \in MemberProfile, \quad m.user\_id \neq \text{NULL} \implies \exists u \in Users \text{ sao cho } m.user\_id = u.user\_id$$
    Chính sách hành vi khi xóa người dùng: \texttt{ON DELETE SET NULL}.
    
    \item \textbf{C3.2 (Parent-Child Reference)}:
    $$\forall p \in ParentChildRelation, \quad \exists m_1, m_2 \in MemberProfile \text{ sao cho } p.parent\_id = m_1.member\_id \land p.child\_id = m_2.member\_id$$
    Chính sách hành vi khi xóa thành viên: \texttt{ON DELETE CASCADE}.
\end{itemize}

\subsubsection{Ràng buộc Nghiệp vụ Phức tạp liên Quan hệ (Cross-Table \& Semantic Constraints)}

\begin{definition}[\textbf{Ràng buộc C4.1: Ngày sinh - Ngày mất và Vòng đời}]
Với mọi thành viên $m \in MemberProfile$:
$$\text{Nếu } m.is\_alive = \text{\texttt{FALSE}} \implies m.death\_date \ge m.birth\_date$$
$$\text{Nếu } m.is\_alive = \text{\texttt{TRUE}} \implies m.death\_date \text{ IS NULL}$$
\end{definition}

\begin{definition}[\textbf{Ràng buộc C4.2: Ngăn ngừa Vòng lặp Quan hệ Huyết thống (Acyclic Genealogic Constraint)}]
Đồ thị quan hệ cha mẹ - con cái $G = (V, E)$ trong đó $V = \{member\_id\}$ và $E = \{(parent\_id, child\_id)\}$ bắt buộc phải là một \textbf{Đồ thị có hướng không chu trình (Directed Acyclic Graph - DAG)}.
$$\forall m \in MemberProfile, \quad (m, m) \notin E^+ \quad (\text{Không một thành viên nào có thể là tổ tiên của chính mình})$$
\end{definition}

\begin{definition}[\textbf{Ràng buộc C4.3: Tuổi của Cha/Mẹ khi sinh Con}]
Với mọi cặp quan hệ $(p, c) \in ParentChildRelation$ liên kết với hồ sơ thành viên $m_{parent}$ và $m_{child}$:
$$m_{child}.birth\_date - m_{parent}.birth\_date \ge 12 \text{ (năm)}$$
\end{definition}

\subsection{Tóm tắt Phần thiết kế}
Trong phần này, bài báo cáo đã hoàn thành việc chuyển đổi mô hình nghiệp vụ của dự án \textbf{FamilyConnect} sang \textbf{Lược đồ Logic chuẩn hóa}. Bằng việc áp dụng lý thuyết Phụ thuộc hàm và các thuật toán phân rã, lược đồ dữ liệu đã đạt dạng chuẩn BCNF/3NF, đảm bảo loại bỏ tình trạng dư thừa và các bất thường dữ liệu. Hệ thống các ràng buộc toàn vẹn mức logic được thiết lập chặt chẽ sẽ là tiền đề trực tiếp để tiến hành \textbf{Thiết kế Vật lý và Viết script DDL} ở các bước tiếp theo của dự án.
