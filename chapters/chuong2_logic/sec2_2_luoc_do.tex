\section{Danh sách các lược đồ quan hệ (Relational Schemas)}
\label{sec:2_2_luoc_do_quan_he}

\subsection{Lược đồ các bảng phân hệ Người dùng \& Bảo mật}
\label{subsec:2_2_1_nguoidung_baomat}

% Danh sách bảng: Users, Roles, Permissions, UserRoles, RolePermissions, AuditLogs
% Phân hệ này hiện thực hóa yêu cầu "Đăng ký và xác thực người dùng" và
% "Kiểm soát truy cập dựa trên vai trò (RBAC)" trong đề xuất dự án.
% Users - Roles là quan hệ n-n (một người dùng có thể giữ nhiều vai trò,
% một vai trò áp dụng cho nhiều người dùng) -> tách thành bảng liên kết
% UserRoles với khóa chính kép.
% Roles - Permissions cũng là quan hệ n-n (một vai trò có nhiều quyền,
% một quyền thuộc nhiều vai trò) -> tách thành bảng liên kết
% RolePermissions với khóa chính kép.
% AuditLogs ghi lại hành động của Users theo quan hệ 1-n (một người dùng
% có nhiều bản ghi audit log).

\begin{itemize}
  \item \textbf{Users}(\underline{user\_id}, username, email, password\_hash, full\_name, status, created\_at)
  \item \textbf{Roles}(\underline{role\_id}, role\_name, description)
  \item \textbf{Permissions}(\underline{permission\_id}, permission\_name, description)
  \item \textbf{UserRoles}(\underline{user\_id} (FK $\rightarrow$ Users), \underline{role\_id} (FK $\rightarrow$ Roles), assigned\_at)
  \item \textbf{RolePermissions}(\underline{role\_id} (FK $\rightarrow$ Roles), \underline{permission\_id} (FK $\rightarrow$ Permissions))
  \item \textbf{AuditLogs}(\underline{log\_id}, user\_id (FK $\rightarrow$ Users), action, target\_table, target\_id, created\_at)
\end{itemize}

\subsection{Lược đồ các bảng phân hệ Gia tộc \& Phả hệ}
\label{subsec:2_2_2_giatoc_phahe}

% Danh sách bảng: Clans, Branches, Members, Marriages, CareerProfiles
% Ánh xạ từ nhóm chức năng "Quản lý Gia phả" và "Danh bạ & Di sản".
% Clans (1) - Branches (n): mỗi dòng họ gồm nhiều chi/nhánh/phái.
% Branches (1) - Members (n): mỗi chi/nhánh gồm nhiều thành viên.
% Members - Members: quan hệ đệ quy cha/mẹ - con qua parent_id (tự tham
% chiếu về chính bảng Members).
% Members - Members qua Marriages: quan hệ n-n (một người có thể có
% nhiều cuộc hôn nhân theo thời gian) -> tách thành bảng liên kết
% Marriages với 2 khóa ngoại độc lập trỏ về Members.
% Members (1) - CareerProfiles (n): một thành viên có thể có nhiều hồ sơ
% học vấn/nghề nghiệp theo từng giai đoạn.

\begin{itemize}
  \item \textbf{Clans}(\underline{clan\_id}, clan\_name, description)
  \item \textbf{Branches}(\underline{branch\_id}, clan\_id (FK $\rightarrow$ Clans), branch\_name, description)
  \item \textbf{Members}(\underline{member\_id}, branch\_id (FK $\rightarrow$ Branches), user\_id (FK $\rightarrow$ Users, nullable), parent\_id (FK $\rightarrow$ Members, tự tham chiếu, nullable), full\_name, dob, dod, gender)
  \item \textbf{Marriages}(\underline{marriage\_id}, member\_id\_1 (FK $\rightarrow$ Members), member\_id\_2 (FK $\rightarrow$ Members), marriage\_date)
  \item \textbf{CareerProfiles}(\underline{profile\_id}, member\_id (FK $\rightarrow$ Members), education\_level, school\_name, occupation, company, start\_date, end\_date)
\end{itemize}

\subsection{Lược đồ các bảng phân hệ Tương tác, Sự kiện \& Di sản}
\label{subsec:2_2_3_tuongtac_sukien_disan}

% Danh sách bảng: Posts, Comments, Albums, Events, EventParticipants,
% HeritageItems, NotableMembers
% Ánh xạ từ nhóm chức năng "Cộng đồng & Tương tác" và "Quản lý Sự kiện".
% Posts (1) - Comments (n): một bài đăng nhận nhiều bình luận.
% Events (1) - Albums (n): thư viện ảnh gắn theo từng sự kiện (nullable
% nếu album thuộc chung dòng họ, không gắn sự kiện cụ thể).
% Events - Users: quan hệ n-n qua bảng liên kết EventParticipants
% (chức năng RSVP), khóa chính kép (event_id, user_id).
% HeritageItems: tài liệu/di sản số gắn với Clans theo quan hệ 1-n.
% NotableMembers: bảng mở rộng thuộc tính cho thành viên tiêu biểu,
% quan hệ 1-1 tùy chọn với Members (chuyên biệt hóa một phần).

\begin{itemize}
  \item \textbf{Posts}(\underline{post\_id}, user\_id (FK $\rightarrow$ Users), clan\_id (FK $\rightarrow$ Clans), content, created\_at)
  \item \textbf{Comments}(\underline{comment\_id}, post\_id (FK $\rightarrow$ Posts), user\_id (FK $\rightarrow$ Users), content, created\_at)
  \item \textbf{Albums}(\underline{album\_id}, event\_id (FK $\rightarrow$ Events, nullable), clan\_id (FK $\rightarrow$ Clans), title, created\_at)
  \item \textbf{Events}(\underline{event\_id}, clan\_id (FK $\rightarrow$ Clans), name, event\_date, location, created\_by (FK $\rightarrow$ Users))
  \item \textbf{EventParticipants}(\underline{event\_id} (FK $\rightarrow$ Events), \underline{user\_id} (FK $\rightarrow$ Users), status)
  \item \textbf{HeritageItems}(\underline{item\_id}, clan\_id (FK $\rightarrow$ Clans), title, description, file\_url, created\_at)
  \item \textbf{NotableMembers}(\underline{notable\_id}, member\_id (FK $\rightarrow$ Members), achievement, description)
\end{itemize}

\subsection{Lược đồ các bảng phân hệ Thông báo (Notifications)}
\label{subsec:2_2_4_thongbao}

% Danh sách bảng: Notifications, NotificationSettings
% Ánh xạ từ chức năng "Nhắc lịch giỗ/sự kiện" và "Thông báo tương tác mới".
% Users (1) - Notifications (n): một người dùng nhận nhiều thông báo.
% NotificationSettings lưu tùy chọn bật/tắt từng loại thông báo theo
% từng người dùng -> quan hệ n-n giữa Users và "loại thông báo",
% hiện thực qua bảng có khóa chính kép (user_id, notification_type).

\begin{itemize}
  \item \textbf{Notifications}(\underline{notification\_id}, user\_id (FK $\rightarrow$ Users), type, content, related\_table, related\_id, is\_read, created\_at)
  \item \textbf{NotificationSettings}(\underline{user\_id} (FK $\rightarrow$ Users), \underline{notification\_type}, is\_enabled)
\end{itemize}

\subsection{Lược đồ bảng phân hệ Tệp tin dùng chung (Media)}
\label{subsec:2_2_5_media}

% Danh sách bảng: MediaFiles
% Bảng dùng chung để lưu trữ tệp tin (ảnh, video, tài liệu) có thể đính
% kèm cho nhiều loại đối tượng khác nhau (Posts, Comments, Members,...).
% Đây là quan hệ đa hình (polymorphic association): related_table xác
% định bảng đích, related_id xác định bản ghi cụ thể trong bảng đó,
% thay vì tạo khóa ngoại riêng cho từng bảng liên quan.

\begin{itemize}
  \item \textbf{MediaFiles}(\underline{media\_id}, file\_url, file\_type, file\_size, uploaded\_by (FK $\rightarrow$ Users), related\_table, related\_id, created\_at)
\end{itemize}

\subsection{Lược đồ các bảng phân hệ Quỹ \& Đóng góp (Donation)}
\label{subsec:2_2_6_quy_donggop}

% Danh sách bảng: Funds, Donations, Expenses
% Ánh xạ từ chức năng "Quản lý quỹ họ tộc / đóng góp / chi tiêu".
% Clans (1) - Funds (n): một dòng họ có thể quản lý nhiều quỹ (quỹ
% chung, quỹ xây từ đường, quỹ khuyến học,...).
% Funds (1) - Donations (n): một quỹ nhận nhiều khoản đóng góp.
% Members (1) - Donations (n): một thành viên có thể đóng góp nhiều lần.
% Funds (1) - Expenses (n): một quỹ phát sinh nhiều khoản chi.

\begin{itemize}
  \item \textbf{Funds}(\underline{fund\_id}, clan\_id (FK $\rightarrow$ Clans), fund\_name, balance, description, created\_at)
  \item \textbf{Donations}(\underline{donation\_id}, fund\_id (FK $\rightarrow$ Funds), member\_id (FK $\rightarrow$ Members), amount, donation\_date, note)
  \item \textbf{Expenses}(\underline{expense\_id}, fund\_id (FK $\rightarrow$ Funds), amount, expense\_date, description, created\_by (FK $\rightarrow$ Users))
\end{itemize}

\subsection{Lược đồ bảng phân hệ Cấu hình hệ thống (Settings)}
\label{subsec:2_2_7_caihinh}

% Danh sách bảng: SystemSettings, ClanSettings
% SystemSettings: cấu hình chung áp dụng toàn hệ thống (ví dụ: giới hạn
% dung lượng upload, ngôn ngữ mặc định), lưu dạng key-value.
% Clans (1) - ClanSettings (n): mỗi dòng họ có thể tùy chỉnh cấu hình
% riêng (ví dụ: quyền riêng tư gia phả, giao diện trang họ), cũng lưu
% dạng key-value với khóa chính kép (clan_id, setting_key).

\begin{itemize}
  \item \textbf{SystemSettings}(\underline{setting\_key}, setting\_value, description, updated\_at)
  \item \textbf{ClanSettings}(\underline{clan\_id} (FK $\rightarrow$ Clans), \underline{setting\_key}, setting\_value)
\end{itemize}
