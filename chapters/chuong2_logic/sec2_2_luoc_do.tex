\section{Lược đồ Quan hệ Logic (Relational Schemas)}
\label{sec:2_2_luoc_do_quan_he}

Sau khi áp dụng các quy tắc chuyển đổi từ mô hình Thực thể -- Liên kết mở rộng (ERD/EERD), hệ thống \textbf{FamilyConnect} được cụ thể hóa thành tập hợp 13 lược đồ quan hệ (Relation Schemas) phân bổ vào 5 phân hệ nghiệp vụ chính.

\subsection{Phân hệ Người dùng \& Bảo mật (User \& Security)}
\label{subsec:2_2_1_user_security}

\begin{enumerate}[label=\textbf{\arabic*.}]
    \item \textbf{Users} (\underline{user\_id}, username, email, password\_hash, full\_name,
          phone\_number, created\_at, is\_active, role\_code)\\
          \textit{Ánh xạ từ:} Thực thể \texttt{NGUOI\_DUNG} (quan hệ 1--1 với MemberProfile).

    \item \textbf{AuditLog} (\underline{audit\_id}, user\_id\textsuperscript{FK},
          action\_type, entity\_name, entity\_id, timestamp, ip\_address)\\
          \textit{Ánh xạ từ:} Thực thể yếu \texttt{NHAT\_KY\_KIEM\_TOAN} (phụ thuộc vào Users).
\end{enumerate}

\subsection{Phân hệ Gia đình \& Gia phả (Family \& Genealogy)}
\label{subsec:2_2_2_family_genealogy}

\begin{enumerate}[resume, label=\textbf{\arabic*.}]
    \item \textbf{FamilyTree} (\underline{family\_id}, family\_name, founding\_date,
          description, created\_by\_user\_id\textsuperscript{FK})\\
          \textit{Ánh xạ từ:} Thực thể \texttt{DONG\_HO}.

    \item \textbf{Branch} (\underline{branch\_id}, branch\_name, generation\_level,
          description, family\_id\textsuperscript{FK})\\
          \textit{Ánh xạ từ:} Thực thể \texttt{CHI\_TOC} (quan hệ 1--N từ FamilyTree).

    \item \textbf{MemberProfile} (\underline{member\_id}, first\_name, last\_name, gender,
          birth\_date, death\_date, is\_alive, bio, occupation, education\_level,
          current\_location, user\_id\textsuperscript{FK}, branch\_id\textsuperscript{FK})\\
          \textit{Ánh xạ từ:} Thực thể \texttt{THANH\_VIEN} (quan hệ 1--1 với Users, N--1 với Branch).

    \item \textbf{ParentChildRelation} (\underline{parent\_id\textsuperscript{FK},
          child\_id\textsuperscript{FK}}, relation\_type)\\
          \textit{Ánh xạ từ:} Mối quan hệ đệ quy \texttt{CHA\_ME\_CON} (N--M đệ quy trên MemberProfile).

    \item \textbf{MarriageRelation} (\underline{marriage\_id}, husband\_id\textsuperscript{FK},
          wife\_id\textsuperscript{FK}, marriage\_date, divorce\_date, status)\\
          \textit{Ánh xạ từ:} Mối quan hệ đệ quy \texttt{HON\_NHAN} (dạng lịch sử, N--M theo thời gian).
\end{enumerate}

\subsection{Phân hệ Cộng đồng \& Sự kiện (Community \& Events)}
\label{subsec:2_2_3_community_events}

\begin{enumerate}[resume, label=\textbf{\arabic*.}]
    \item \textbf{Post} (\underline{post\_id}, title, content, visibility\_scope, created\_at,
          author\_member\_id\textsuperscript{FK}, family\_id\textsuperscript{FK})\\
          \textit{Ánh xạ từ:} Thực thể \texttt{BAI\_VIET} (quan hệ N--1 với MemberProfile và FamilyTree).

    \item \textbf{PostInteraction} (\underline{interaction\_id}, post\_id\textsuperscript{FK},
          member\_id\textsuperscript{FK}, reaction\_type, comment\_text, created\_at)\\
          \textit{Ánh xạ từ:} Mối quan hệ N--M \texttt{TUONG\_TAC} (MemberProfile $\leftrightarrow$ Post).

    \item \textbf{Event} (\underline{event\_id}, event\_title, event\_description, start\_time,
          end\_time, location, event\_type, family\_id\textsuperscript{FK},
          organizer\_member\_id\textsuperscript{FK})\\
          \textit{Ánh xạ từ:} Thực thể \texttt{SU\_KIEN}.

    \item \textbf{EventRSVP} (\underline{event\_id\textsuperscript{FK},
          member\_id\textsuperscript{FK}}, rsvp\_status, updated\_at)\\
          \textit{Ánh xạ từ:} Mối quan hệ N--M \texttt{THAM\_GIA} (MemberProfile $\leftrightarrow$ Event).
\end{enumerate}

\subsection{Phân hệ Di sản \& AI (Heritage \& AI Services)}
\label{subsec:2_2_4_heritage_ai}

\begin{enumerate}[resume, label=\textbf{\arabic*.}]
    \item \textbf{HeritageDocument} (\underline{document\_id}, title, document\_type,
          file\_path, historical\_period, summary,
          uploaded\_by\_member\_id\textsuperscript{FK}, family\_id\textsuperscript{FK})\\
          \textit{Ánh xạ từ:} Thực thể \texttt{TAI\_LIEU\_DI\_SAN}.

    \item \textbf{AIAssistantLog} (\underline{log\_id}, member\_id\textsuperscript{FK},
          prompt\_query, response\_summary, vector\_embedding\_id, logged\_at)\\
          \textit{Ánh xạ từ:} Thực thể yếu \texttt{NHAT\_KY\_TRO\_LY\_AI}.
\end{enumerate}
